//SAVING
function saveGame() {

	const saveData = {
		staples: staples,
		autoStaplers: autoStaplers,
		autoStaplerPrice: autoStaplerPrice,
		hasAlphaDrone: hasAlphaDrone,
		hasBetaDrone: hasBetaDrone,
		hasGammaDrone: hasGammaDrone,
		gammaDroneActive: gammaDroneActive
	};

	const saveString = JSON.stringify(saveData);
	const blob = new Blob([saveString], { type: "text/plain" });
	const url = URL.createObjectURL(blob);

	const a = document.createElement("a");
	a.href = url;
	a.download = "galactic_staples_employee_25.12.0000_neurochip_save.txt";
	a.click();

	URL.revokeObjectURL(url);
}


//LOADING
function loadGame(event) {
	const file = event.target.files[0];
	if (!file) return;

	const reader = new FileReader();
	reader.onload = function(e) {
		try {
			const saveData = JSON.parse(e.target.result);

			staples = saveData.staples || 0;
			autoStaplers = saveData.autoStaplers || 0;
			autoStaplerPrice = saveData.autoStaplerPrice || 10;
			hasAlphaDrone = saveData.hasAlphaDrone || false;
			hasBetaDrone = saveData.hasBetaDrone || false;
			hasGammaDrone = saveData.hasGammaDrone || false;
			gammaDroneActive = saveData.gammaDroneActive ?? true;

			updateUI();
			renderDrones();

		} catch (error) {
			alert("Invalid save file. Are you sure you have a company issued NeuroChip™ memory file?");
		}
	};
	reader.readAsText(file);
}

//PRIMARY FUNCTIONS
function updateUI() {
	counterEl.textContent = Math.floor(staples).toLocaleString();
	let displayPrice = daysArray[dayIndex] === "Saturday" ? Math.floor(autoStaplerPrice * 0.75) : autoStaplerPrice;
	priceEl.textContent = displayPrice.toLocaleString();
	autoCounterEl.textContent = autoStaplers.toLocaleString();
	document.getElementById("spsCounter").textContent = measuredSPS.toLocaleString();
}
function makeStaple() {
	let amount = hasAlphaDrone ? 1.5 : 1;
	staples += amount;
	updateUI();
}
function buyAutoStapler() {
    let currentPrice = daysArray[dayIndex] === "Saturday" ? Math.floor(autoStaplerPrice * 0.5) : autoStaplerPrice;
    if (staples >= currentPrice) {
        staples -= currentPrice;
        let staplerAmount = daysArray[dayIndex] === "Wednesday" ? 2 : 1;
        autoStaplers += staplerAmount;
        autoStaplerPrice = Math.ceil(autoStaplerPrice * 1.25);
        updateUI();
    }
}

function manualMakeStaple() {
	if (daysArray[dayIndex] === "Sunday") {
		alert("All staple production is halted on Sunday. This is to optimize employee production on Monday.");
		return;
	}
	let clickValue = hasBetaDrone ? 3 : 1;
	staples += clickValue;
	manualClicks += 1;
	updateUI();


//DOWNLADING THE NOTE
	if (manualClicks === 1000 && !noteTrigger) {
		noteTrigger = true;
		downloadNote();
	}
}

    if (staples >= alphaCost && !hasAlphaDrone) {
